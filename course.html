true_cost_analysis.py 
def true_cost_analysis(trades_per_day, avg_size, fee=0.0004):
    # fee: Binance Futures avg maker+taker ~0.04%
    monthly = trades_per_day * avg_size * 2 * fee * 30
    annual_pct = monthly * 12 / 500 * 100
    return {"monthly_usd": monthly, "annual_pct_on_500": annual_pct}

# До оптимизации: 159 сделок/день × $22
# → $124/мес → 24.9% годовых только на комиссиях
result = true_cost_analysis(159, 22)
print(result)
